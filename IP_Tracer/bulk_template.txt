// ─── BULK IP REPORT ──────────────────────────────────
// Each item = one IP from the bulk response
const allItems = $input.all();
const chatId = (allItems.length > 0 && allItems[0].json.chatId) || 0;

let msg = '';
msg += '╔══════════════════════════════════════╗\n';
msg += '║   📊 *BULK IP INTELLIGENCE REPORT*  ║\n';
msg += '╚══════════════════════════════════════╝\n\n';

// ─── PARSE ──────────────────────────────────────
const results = [];
const errors = [];

for (const item of allItems) {
  const r = item.json;
  // Handle both raw JSON and stringified data
  let data = r;
  if (r.data && typeof r.data === 'string') {
    try {
      const parsed = JSON.parse(r.data);
      if (Array.isArray(parsed)) {
        parsed.forEach(p => results.push(p));
        continue;
      } else if (parsed.ip) {
        results.push(parsed);
        continue;
      }
    } catch (e) { /* not JSON */ }
  }
  if (r.ip) results.push(r);
  else if (r.error || r.success === false) errors.push(r);
}

if (results.length === 0) {
  msg += '⚠️ No valid IP data returned from the API.\n';
  return [{ json: { chatId, message: msg, parse_mode: 'Markdown' } }];
}

if (errors.length > 0) {
  msg += '⚠️ ' + errors.length + ' IP(s) returned API errors. Results may be incomplete.\n\n';
}

msg += '━━ 📋 *IP COMPARISON TABLE* ━━\n\n';

// Helper to format a single IP entry
function describeIP(d) {
  const loc = d.location || {};
  const conn = d.connection || {};
  const tz = d.time_zone || {};
  const flag = loc.country_flag_emoji || '';
  const country = d.country_name || d.country_code || 'Unknown';
  const city = d.city || '';
  const region = d.region_name || '';
  const isp = conn.isp || 'Unknown';
  const asn = conn.asn ? 'AS' + conn.asn : '';
  const org = conn.organization_type || '';
  const connType = d.connection_type || '';
  const routing = d.ip_routing_type || '';
  const lat = d.latitude;
  const lon = d.longitude;

  // Inline risk score
  const sec = d.security || {};
  let risk = 0;
  if (sec.is_proxy) risk += 40;
  if (sec.is_tor) risk += 50;
  if (sec.is_crawler) risk += 20;
  if (sec.is_vpn) risk += 30;
  if (sec.threat_level === 'medium') risk += 30;
  if (sec.threat_level === 'high') risk += 70;

  risk = Math.min(100, Math.max(0, risk));
  const rEmoji = risk <= 15 ? '🟢' : risk <= 50 ? '🟡' : '🔴';
  const rLabel = risk <= 15 ? 'Low' : risk <= 50 ? 'Medium' : 'High';

  // Hemisphere
  const hemi = lat != null ? (lat >= 0 ? 'Northern' : 'Southern') + ', ' + (lon >= 0 ? 'Eastern' : 'Western') : '';
  const coords = lat != null && lon != null ? lat + ', ' + lon : '';

  // Location Precision
  const prec = (d.city && d.zip) ? '📌 High (City+ZIP)' : d.city ? '📍 Medium (City)' : '🌐 Low (Country)';

  // ISP Category / Hosting badge
  const isCDN = isp.indexOf('Cloudflare') >= 0 || isp.indexOf('Vultr') >= 0 || isp.indexOf('AWS') >= 0 || isp.indexOf('Amazon') >= 0 || isp.indexOf('DigitalOcean') >= 0 || isp.indexOf('Azure') >= 0;
  const isHosting = org.indexOf('Internet Hosting') >= 0;
  const isMobile = connType === 'mobile wireless' || connType === 'mobile';
  const typeBadge = isCDN ? '🌐 CDN/Proxy' : isHosting ? '☁️ Datacenter' : isMobile ? '📱 Mobile' : (d.connection && d.connection.home) ? '🏠 Residential' : '🔌 General';
  const useCase = isHosting ? '🖥️ Web Hosting/API' : isCDN ? '☁️ CDN/VPS' : isMobile ? '📱 End-User' : (d.connection && d.connection.home) ? '🏠 Personal' : connType === 'tx' ? '☁️ Server/VPS' : '💻 General';
  const trust = isCDN ? '🔴 Low (Proxy/CDN)' : isHosting ? '🟡 Medium (DC)' : isMobile ? '🟢 High (Mobile)' : '🟡 Medium';
  // Time mood
  let mood = '';
  if (tz.current_time) {
    try {
      const h = parseInt(tz.current_time.substring(11,13));
      if (h < 6) mood = '🌙 Night';
      else if (h < 12) mood = '☀️ Morning';
      else if (h < 17) mood = '🌤️ Afternoon';
      else if (h < 21) mood = '🌆 Evening';
      else mood = '🌃 Late';
    } catch(e) { mood = ''; }
  }

  let entry = '';
  entry += '━━ 🌐 ' + flag + ' *' + d.ip + '* ━━\n';
  entry += '📍 ' + city + (city && region ? ', ' : '') + region + (country ? ', ' : '') + country + '\n';
  entry += '🏢 ' + isp;
  if (asn) entry += ' (' + asn + ')';
  entry += '\n';
  if (org) entry += '🏭 ' + org + '\n';
  if (connType || routing) entry += '🔌 ' + connType + (connType && routing ? ' / ' : '') + routing + '\n';
  if (coords) entry += '🌐 ' + coords + '\n';
  if (hemi) entry += '🧭 ' + hemi + '\n';
  if (mood) entry += '🕐 ' + mood + ' (' + (tz.current_time || '') + ')\n';
  entry += '🛡 Risk: ' + rEmoji + ' ' + rLabel + ' (' + risk + '/100)\n';
  entry += '🎯 ' + prec + ' | ' + typeBadge + '\n';
  entry += '🛡 Trust: ' + trust + ' | ' + useCase + '\n';
  entry += '\n';

  return entry;
}

// Generate individual entries (compact)
for (let i = 0; i < results.length; i++) {
  const d = results[i];
  msg += describeIP(d);

  // Break long messages — Telegram limit ~4000 chars
  if (msg.length > 3500 && i < results.length - 1) {
    msg += '... *' + (results.length - i - 1) + ' more IP(s)*\n\n';
    break;
  }
}

// ─── EXECUTIVE SUMMARY ──────────────────────────
msg += '━━ 📊 *EXECUTIVE SUMMARY* ━━\n\n';

const uniqueCountries = [...new Set(results.map(d => d.country_name).filter(Boolean))];
const uniqueISPs = [...new Set(results.map(d => (d.connection || {}).isp).filter(Boolean))];
const avgRisk = results.reduce((s, d) => {
  const sec = d.security || {};
  let r = 0;
  if (sec.is_proxy) r += 40;
  if (sec.is_tor) r += 50;
  if (sec.is_crawler) r += 20;
  if (sec.is_vpn) r += 30;
  if (sec.threat_level === 'medium') r += 30;
  if (sec.threat_level === 'high') r += 70;
  return s + Math.min(100, Math.max(0, r));
}, 0) / results.length;

const dcIps = results.filter(d => (d.connection || {}).isp === 'Cloudflare' || (d.connection || {}).isp === 'The Constant Company LLC').length;
const mobileIps = results.filter(d => d.connection_type === 'mobile wireless').length;

msg += '📦 *Total IPs:* ' + results.length + '\n';
msg += '🌍 *Countries:* ' + (uniqueCountries.length > 0 ? uniqueCountries.join(', ') : 'N/A') + '\n';
msg += '🏢 *ISPs:* ' + (uniqueISPs.length > 0 ? uniqueISPs.join(', ') : 'N/A') + '\n';
msg += '📈 *Avg Risk:* ' + avgRisk.toFixed(0) + '/100 — ' + (avgRisk <= 15 ? '🟢 Low' : avgRisk <= 50 ? '🟡 Medium' : '🔴 High') + '\n';
if (dcIps > 0) msg += '☁️ *Datacenter IPs:* ' + dcIps + '\n';
if (mobileIps > 0) msg += '📱 *Mobile IPs:* ' + mobileIps + '\n';
msg += '\n';

msg += '💡 *Recommendation:* ';
if (avgRisk <= 15) msg += 'No significant security concerns across the IP set.';
else if (avgRisk <= 50) msg += 'Some IPs show elevated risk. Review flagged entries.';
else msg += '⚠️ Multiple high-risk IPs detected. Immediate review recommended.';
msg += '\n\n';

// ─── NOTES ──────────────────────────────────────
msg += '━━ ℹ️ *NOTES* ━━\n';
msg += '✅ Data from IPStack API.\n';
msg += '🧠 Risk scores calculated inline from available data.\n';
msg += '⚠️ Geolocation is approximate.\n';
msg += '\n';
msg += '━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━\n';
msg += '🤖 *TracedIP Intelligence Bot*\n';
msg += '📡 [ipstack.com](https://ipstack.com)\n';

return [{ json: { chatId, message: msg, parse_mode: 'Markdown' } }];
